Payload format & verification

Headers

Content-Type: application/json
X-Snitchr-Signature: sha256=<hex>
X-Snitchr-Timestamp: <unix-seconds>

Body

{
  "id": "evt_a1b2c3d4e5f6a7b8",
  "event": "alert | recovery | down | test",
  "timestamp": "2026-04-01T12:00:00.000Z",
  "machine": { "id": "web-1", "label": "Production Web" },
  "level": 2,
  "levelName": "WARN",
  "title": "WARN: Production Web",
  "body": "SSH login from 1.2.3.4",
  "priority": "high"
}

Verify (Node.js)

import { createHmac, timingSafeEqual } from 'crypto';

const ts = req.headers['x-snitchr-timestamp'];
const sig = req.headers['x-snitchr-signature'];
if (Math.abs(Date.now() / 1000 - Number(ts)) > 300)
  throw new Error('Timestamp too old');

const expected = 'sha256=' + createHmac('sha256', SECRET)
  .update(ts + '.' + rawBody).digest('hex');
if (!timingSafeEqual(Buffer.from(sig), Buffer.from(expected)))
  throw new Error('Invalid signature');
